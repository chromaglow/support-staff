# Support Fins Desktop Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A Windows Electron app that runs the upstream Support Fins web app unmodified from a git submodule, saves exports through native dialogs, and adds a one-click "Send to Bambu" that writes the finned 3MF and opens it in Bambu Studio.

**Architecture:** Electron main process serves `upstream/web/` over a custom `app://` scheme (with a strict CSP), injects one shim script into the served HTML, and exposes a narrow IPC bridge through a sandboxed preload. The shim imports the same four upstream functions that upstream's own Export 3MF uses, so the Send to Bambu file is identical to upstream's export. Nothing under `upstream/` is ever edited.

**Tech Stack:** Node 24, Electron 44, electron-builder 26, @playwright/test 1.64 (Electron driver), `node --test` for unit tests, Deno 2 for upstream's own suite. Main process and tests are CommonJS; the shim is an ES module because it imports upstream ES modules.

**Spec:** `docs/specs/2026-10-10-support-fins-desktop-design.md`. Read it first.

**Repo:** `C:\dev\GitHub\support staff`, remote `https://github.com/chromaglow/support-staff`, branch `main`. The submodule `upstream/` points at `https://github.com/gittrahan/support-fins`. Work on `main` directly; commit after every task; push at the end of each phase.

```js
#!/usr/bin/env node
// Research: run upstream's CLI over every .stl/.3mf in a folder and print a Markdown
// table: walls, tines, unserved, floating, seconds. Usage:
//   node scripts/survey.js <folder-or-file>... [--rot x,y,z] [--out <dir>]
const { spawnSync } = require('child_process');
const fs = require('fs');
const os = require('os');
const path = require('path');

const ROOT = path.join(__dirname, '..');
const CLI = path.join(ROOT, 'upstream', 'plugins', 'cli', 'support-fins.js');

const args = process.argv.slice(2);
const rotIdx = args.indexOf('--rot');
const rot = rotIdx >= 0 ? args.splice(rotIdx, 2)[1] : null;
const outIdx = args.indexOf('--out');
const outDir = outIdx >= 0 ? args.splice(outIdx, 2)[1] : fs.mkdtempSync(path.join(os.tmpdir(), 'survey-'));
fs.mkdirSync(outDir, { recursive: true });

const files = args.flatMap((a) => fs.statSync(a).isDirectory()
  ? fs.readdirSync(a).filter((f) => /\.(stl|3mf)$/i.test(f)).map((f) => path.join(a, f))
  : [a]);
if (!files.length) { console.error('no .stl/.3mf inputs'); process.exit(2); }

console.log(`| file | walls | tines | unserved | floating | s | note |\n|---|---|---|---|---|---|---|`);
for (const f of files) {
  const out = path.join(outDir, path.basename(f).replace(/\.(stl|3mf)$/i, '') + '-fins.3mf');
  const cliArgs = [CLI, f, '--json', '-o', out, ...(rot ? ['--rot', rot] : [])];
  const t0 = Date.now();
  const r = spawnSync(process.execPath, cliArgs, { encoding: 'utf8' });
  const s = ((Date.now() - t0) / 1000).toFixed(1);
  const line = (r.stdout || '').trim().split('\n').pop() || '';
  let row;
  try {
    const j = JSON.parse(line);
    const st = j.stats || {};
    row = `| ${path.basename(f)} | ${st.braces ?? ''} | ${st.tines ?? ''} | ${st.unserved ?? ''} | ${st.floating ?? ''} | ${s} | ${j.report ?? ''} |`;
  } catch {
    row = `| ${path.basename(f)} | | | | | ${s} | ERROR: ${(r.stderr || line).trim().split('\n').pop()} |`;
  }
  console.log(row);
}
console.error(`outputs in ${outDir}`);
```

```js
// Rolling log file: one line per entry, rotated once past MAX bytes.
const fs = require('fs');
const path = require('path');

const MAX = 2 * 1024 * 1024;

function createLogger(file) {
  fs.mkdirSync(path.dirname(file), { recursive: true });
  function write(level, msg) {
    const line = `${new Date().toISOString()} ${level.toUpperCase()} ${msg}\n`;
    try {
      let size = 0;
      try { size = fs.statSync(file).size; } catch { /* no file yet */ }
      if (size > MAX) { try { fs.renameSync(file, `${file}.1`); } catch { /* keep writing */ } }
      fs.appendFileSync(file, line);
    } catch { /* logging never throws */ }
  }
  return {
    file,
    info: (m) => write('info', m),
    warn: (m) => write('warn', m),
    error: (m) => write('error', m),
  };
}

module.exports = { createLogger, MAX };
```

```js
// Filenames and writes for the output folder. Pure except writeAtomic.
const fs = require('fs');
const path = require('path');

/** A part name the file system and a slicer both accept. Never empty. */
function safeBase(name) {
  const s = String(name ?? '')
    .replace(/[^A-Za-z0-9._ -]/g, '')
    .trim()
    .slice(0, 80)
    .replace(/^[. ]+|[. ]+$/g, '');
  return s || 'part';
}

/** `<dir>/<base><ext>`, or `<base>-2<ext>`, `-3`, ... while taken. */
function uniquePath(dir, base, ext, exists = fs.existsSync) {
  let p = path.join(dir, `${base}${ext}`);
  for (let n = 2; exists(p); n++) {
    if (n > 999) throw new Error(`no free name for ${base}${ext} in ${dir}`);
    p = path.join(dir, `${base}-${n}${ext}`);
  }
  return p;
}

/** Write to `<target>.part`, then rename: a crash never leaves a half file at target. */
async function writeAtomic(target, bytes) {
  await fs.promises.mkdir(path.dirname(target), { recursive: true });
  const tmp = `${target}.part`;
  await fs.promises.writeFile(tmp, bytes);
  await fs.promises.rename(tmp, target);
}

module.exports = { safeBase, uniquePath, writeAtomic };
```

```js
  process.env['ProgramFiles(x86)'] && path.join(process.env['ProgramFiles(x86)'], 'Bambu Studio', 'bambu-studio.exe'),

/** `"C:\x\bambu-studio.exe" "%1"` or `C:\x\bambu-studio.exe %1` -> the exe path, else null. */
function exeFromCommand(cmd) {
  if (typeof cmd !== 'string') return null;
  const m = cmd.trim().match(/^"([^"]+\.exe)"|^(\S+\.exe)/i);
  return m ? (m[1] || m[2]) : null;
}

/** Default value of a registry key via `reg query`, or null. */
function regQuery(key) {
  try {
    const out = execFileSync('reg', ['query', key, '/ve'], { encoding: 'utf8', windowsHide: true });
    const line = out.split(/\r?\n/).find((l) => /REG_(SZ|EXPAND_SZ)/.test(l));
    if (!line) return null;
    return line.split(/REG_(?:SZ|EXPAND_SZ)\s+/)[1]?.trim() ?? null;
  } catch {
    return null;
  }
}

function detectBambu({ exists = fs.existsSync, query = regQuery, candidates: list = candidates() } = {}) {
  for (const c of list) if (exists(c)) return c;
  const appPath = query('HKLM\\SOFTWARE\\Microsoft\\Windows\\CurrentVersion\\App Paths\\bambu-studio.exe');
  if (appPath && exists(appPath)) return appPath;
  const open = exeFromCommand(query('HKCR\\bambustudio\\shell\\open\\command'));
  if (open && exists(open)) return open;
  return null;
}

/** Open `file` in Bambu Studio, detached so closing this app leaves the slicer running. */
function launchBambu(exe, file) {
  const child = spawn(exe, [file], { detached: true, stdio: 'ignore' });
  child.unref();
  return child.pid;
}
```

```js
// --- Send to Bambu -----------------------------------------------------------
async function sendToBambu(mods) {
  const { buildExportGeometry } = mods['../ui/export.js'];
  const { writeThreeMF } = mods['../threemf.js'];
  const { splitInterface } = mods['../prop.js'];
  const { sessionOf } = mods['../ui/session.js'];
  const g = buildExportGeometry();
  if (!g) { toast('Load a part and add fins first'); return; }
  // Exactly upstream's export-3mf: part and supports as separate objects, interface
  // crest split when that material is on, the session embedded for reopening.
  const { body, iface } = splitInterface(g.finTris);
  const blob = writeThreeMF(g.partTris, body, g.base, { separate: true, iface, session: sessionOf(g.partTris) });
  const bytes = new Uint8Array(await blob.arrayBuffer());
  const r = await bridge.saveAndOpenInBambu(bytes, g.base);
  if (!r.ok) { toast(`Could not save: ${r.error}`); return; }
  const name = r.path.split(/[\\/]/).pop();
  if (r.opened) toast(`Saved ${name}, opened in Bambu Studio`, { label: 'Show in folder', onclick: () => bridge.showInFolder(r.path) });
  else toast(`Saved ${name}. Bambu Studio not found, set it in settings`, { label: 'Settings', onclick: openSettings });
}

async function showBroken(missing) {

    $('div', { class: 'sfd-row sfd-actions' }, $('button', { type: 'button', class: 'btn', onclick: () => d.close() }, 'OK')));

  d.showModal();
}

function mount(mods, missing) {
  const anchor = document.querySelector('#topbar .menu-wrap') || document.querySelector('#topbar');
  const send = $('button', { class: 'btn', id: 'sfd-send', type: 'button', disabled: '',
    title: 'Save the part and fins as a 3MF in the output folder and open it in Bambu Studio' }, 'Send to Bambu');
  const gear = $('button', { class: 'btn', id: 'sfd-settings-btn', type: 'button', title: 'Support Staff settings',
    'aria-label': 'Settings', onclick: openSettings }, '⚙');
  if (anchor.classList.contains('menu-wrap')) anchor.after(send, gear); else anchor.append(send, gear);
  if (missing.length) {
    send.title = `Send to Bambu is unavailable: ${missing.join('; ')}`;
    bridge.log('error', `startup export check failed: ${missing.join('; ')}`);
    showBroken(missing);
    return;
  }
  send.addEventListener('click', () => sendToBambu(mods).catch((e) => {
    toast(`Send to Bambu failed: ${e.message}`);
    bridge.log('error', e.stack || e.message);
  }));
  // Same readiness rule as upstream's Export menu, polled so upstream stays untouched.
  const partMod = mods['../ui/part.js'];
  setInterval(() => { send.disabled = !(partMod.part && partMod.topology && partMod.lastResult); }, 500);
}

(async () => {

  const { mods, missing } = await loadUpstream();

  bridge.onOpenSettings(openSettings);
```
